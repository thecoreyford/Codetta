67  // Cast was successfull
68  auto& fragments = dynamic_cast<MusiSyncEng::Bar*>(bar)->getFragment();
69 
70  // Cycle through each fragment in a bar.
71  for (int i = 0; i < fragments.size(); ++i)
72  {
73  // timestamp must be offest by previous value
74  float timeStamp = PlaybackSettings::get().getLatestTimestampAddition();
75 
76  auto fragment = fragments[i]->getFirstNoteOfFragment();
77  auto noOfNotes = fragments[i]->getNoOfNotesInFragment();
78  // cycle though all notes in the fragment
79  for (int j = 0; j < noOfNotes; ++j)
80  {
81  // ensure that our offset mapping is correct
82  fragment[j]->changeClef (PlaybackSettings::get().getCurrentClef());
83 
84  //====================================================================================
85  // translate into message on
86  auto messageOn = MidiMessage::noteOn ((int)PlaybackSettings::get().getInstrument(),
87  fragment[j]->getMusicalValue(),
88  (uint8) PlaybackSettings::get().getVelocity());
89  messageOn.setTimeStamp (timeStamp); //< add timestamp
90  MidiEventList::get().addMidiEvent (messageOn); //< add to event list
91 
92  //====================================================================================
93 
94  // calculate the next value
95  timeStamp += PlaybackSettings::get().lengthOfNoteInMillisec (fragment[j]->getValue());
96 
97  //timeStamp += PlaybackSettings::get().getLatestTimestampAddition();
98 
99  // add note off
100  auto messageOff = MidiMessage::noteOff ((int)PlaybackSettings::get().getInstrument(),
101  fragment[j]->getMusicalValue(),
102  (uint8) PlaybackSettings::get().getVelocity());
103  messageOff.setTimeStamp (timeStamp); //< add timestamp
104  MidiEventList::get().addMidiEvent (messageOff); // add to event list
105 
106  // update latest addition

108 
109  }
